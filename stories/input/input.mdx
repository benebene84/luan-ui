# Input

A styled text input. Inside `FormField`, it inherits `disabled`, `required`, and `error`.

## Import

```tsx
import { Input } from "luan-ui";
```

## Usage

<div className="flex max-w-sm flex-col gap-3 py-2">
  <Input placeholder="Enter your email" id="docs-email" type="email" />
  <Input error placeholder="Invalid value" />
  <Input disabled placeholder="Disabled" />
</div>

```tsx
<Label htmlFor="email">Email</Label>
<Input id="email" type="email" placeholder="you@example.com" />
```

### Inside FormField

<div className="max-w-sm py-2">
  <FormField required error>
    <Label htmlFor="docs-input-field">Email</Label>
    <Input id="docs-input-field" type="email" placeholder="you@example.com" />
    <FormHelper>Enter a valid email address.</FormHelper>
  </FormField>
</div>
